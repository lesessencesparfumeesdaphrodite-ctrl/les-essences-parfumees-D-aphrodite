async function notion(env, path, method = "GET", body) {
  for (let i = 0; i < 4; i++) {
    const r = await fetch("https://api.notion.com/v1" + path, {
      method, headers: { Authorization: "Bearer " + env.NOTION_TOKEN, "Notion-Version": NV, "Content-Type": "application/json" },
      body: body ? JSON.stringify(body) : undefined,
    });
    if (r.status === 429) { await new Promise((ok) => setTimeout(ok, 1200 * (i + 1))); continue; }
    const j = await r.json();
    if (!r.ok) throw new Error("Notion " + r.status + " " + (j.message || ""));
    return j;
  }
  throw new Error("Notion trop sollicité");
}

    if (start === 1) break;
  } while (cursor);
  return out;
}

// ---------- Sécurité ----------
async function hmac(env, data) {
  const k = await crypto.subtle.importKey("raw", enc.encode("aphro|" + env.NOTION_TOKEN), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  return b64(await crypto.subtle.sign("HMAC", k, enc.encode(data)));
}
async function hashPwd(pwd, salt) {
  const k = await crypto.subtle.importKey("raw", enc.encode(pwd), "PBKDF2", false, ["deriveBits"]);
  return b64(await crypto.subtle.deriveBits({ name: "PBKDF2", salt: enc.encode(salt), iterations: CFG.PBKDF2, hash: "SHA-256" }, k, 256));
}
const rnd = (n) => b64(crypto.getRandomValues(new Uint8Array(n))).slice(0, n);
async function makeToken(env, id) { const exp = Date.now() + 30 * 864e5; const body = id + "." + exp; return body + "." + await hmac(env, "t|" + body); }
async function readToken(env, req) {
  const m = (req.headers.get("Authorization") || "").match(/^Bearer (.+)$/); if (!m) return null;
  const [id, exp, sig] = m[1].split("."); if (!id || !exp || !sig) return null;
  if (sig !== await hmac(env, "t|" + id + "." + exp) || Date.now() > +exp) return null;
  return id;
}

    const salt = rnd(16); const mdp = salt + "|" + await hashPwd(String(body.mdp), salt);
    const props = { ...eqProps(d), "Empreinte mot de passe (ne pas modifier)": T(mdp), "Statut": { select: { name: "Compte créé" } }, "Compte créé le": { date: { start: maintenant } }, "Dernière connexion": { date: { start: maintenant } } };
    let id;
    if (ex) { await notion(env, "/pages/" + ex.id, "PATCH", { properties: props }); id = ex.id; }
    else { const pg = await notion(env, "/pages", "POST", { parent: { type: "data_source_id", data_source_id: CFG.DS_EQUIPE }, properties: props }); id = pg.id; }
    const base = eqBase(body.base);
    await eqEnvoyer(env, d, "Votre compte est créé", "Votre compte est créé", `<p>Bonjour ${esc(d.prenom)},</p><p>Votre compte est créé. Appuyez sur le bouton pour retrouver votre page et vous identifier avec votre adresse e-mail et votre mot de passe.</p>`, "Ouvrir mon lien", eqLien(base, { ...d, id }));
    return J({ ok: true, jeton: await eqJeton(env, id), profil: eqPublic({ ...d, id }) });
  }
  if (route === "/equipe/connexion") {
    const mail = String(body.mail || "").trim().toLowerCase();
    const c = validMail(mail) ? await eqParMail(env, mail) : null;
    if (!c || !c.mdp || !(await eqMdpOk(c, body.mdp))) return J({ erreur: "Adresse e-mail ou mot de passe incorrect." }, 401);
    await notion(env, "/pages/" + c.id, "PATCH", { properties: { "Dernière connexion": { date: { start: maintenant } } } });
    return J({ ok: true, jeton: await eqJeton(env, c.id), profil: eqPublic(c) });
  }
  if (route === "/equipe/moi") {
    const id = await eqLireJeton(env, body.jeton);
    if (!id) return J({ erreur: "Session expirée : reconnectez-vous." }, 401);
    const c = await eqParId(env, id);
    if (c.archive) return J({ erreur: "Session expirée : reconnectez-vous." }, 401);
    return J({ ok: true, profil: eqPublic(c) });
  }
  if (route === "/equipe/profil-maj") {
    const id = await eqLireJeton(env, body.jeton);
    if (!id) return J({ erreur: "Session expirée : reconnectez-vous." }, 401);
    const c = await eqParId(env, id);
    if (c.archive) return J({ erreur: "Session expirée : reconnectez-vous." }, 401);
    let mail = c.mail; const nm = String(body.mail || "").trim().toLowerCase();
    if (nm && nm !== c.mail) {
      if (!validMail(nm)) return J({ erreur: "Merci d'indiquer une adresse e-mail valide." }, 400);
      if (!(await eqMdpOk(c, body.mdp))) return J({ erreur: "Pour changer votre adresse e-mail, écrivez votre mot de passe actuel." }, 403);
      const autre = await eqParMail(env, nm);
      if (autre && autre.id !== id) return J({ erreur: "Cette adresse e-mail est déjà utilisée par un autre compte." }, 409);
      mail = nm;
    }
    const d = eqDonnees({ ...body, mail });
    if (!d.prenom || !d.nom) return J({ erreur: "Merci d'indiquer votre prénom et votre nom." }, 400);
    for (const k of ["offreDu", "offreAu"]) if (k in body && String(body[k] || "").trim() && !eqDate(body[k])) return J({ erreur: "Écrivez les dates de l'offre ainsi : 01/01/2026." }, 400);
    const cles = EQ_EXTRAS.filter((k) => k in body), x = eqExtras(body);
    for (const k of ["logo", "photo", "photoOffre"]) if (k in body && body[k] && !x[k]) return J({ erreur: "Cette image est trop lourde ou n'est pas reconnue. Essayez-en une autre." }, 400);
    await notion(env, "/pages/" + id, "PATCH", { properties: { ...eqProps(d), ...eqExtrasProps(x, cles) } });
    const apres = await eqParId(env, id);
    return J({ ok: true, profil: eqPublic(apres) });
  }
  if (route === "/equipe/page") {
    const i = String(body.i || "");
    if (!/^[0-9a-f]{8}-?[0-9a-f]{4}-?[0-9a-f]{4}-?[0-9a-f]{4}-?[0-9a-f]{12}$/i.test(i)) return J({ ok: true, vide: true });
    let pg; try { pg = await notion(env, "/pages/" + i); } catch (e) { return J({ ok: true, vide: true }); }
    const ds = pg && pg.parent && (pg.parent.data_source_id || pg.parent.database_id || "");
    if (String(ds).replace(/-/g, "") !== CFG.DS_EQUIPE.replace(/-/g, "")) return J({ ok: true, vide: true });
    if (pg.archived || pg.in_trash) return J({ ok: true, vide: true });
    const c = eqCompte(pg);
    return J({ ok: true, societe: c.societe, logo: c.logo, photo: c.photo, reseaux: c.reseaux, offres: c.offres, photoOffre: c.photoOffre, fidelite: c.fidelite, offreDu: c.offreDu, offreAu: c.offreAu });
  }
  if (route === "/equipe/offres") {
    let lignes = [], cursor, apres = false, vu = false;
    try {
      do {
        const j = await notion(env, `/blocks/${CFG.PAGE_OFFRES_EXH}/children?page_size=100${cursor ? "&start_cursor=" + cursor : ""}`);
        for (const b of j.results) {
          if (b.type === "divider") { apres = true; vu = true; lignes = []; continue; }
          const rt = (b[b.type] && b[b.type].rich_text) || []; const t = rt.map((x) => x.plain_text).join("").trim();
          if (t && ["paragraph", "bulleted_list_item", "numbered_list_item", "heading_1", "heading_2", "heading_3"].includes(b.type) && apres) lignes.push((b.type === "bulleted_list_item" ? "• " : "") + t);
        }
        cursor = j.has_more ? j.next_cursor : null;
      } while (cursor);
    } catch (e) { return J({ ok: true, vide: true }); }
    const texte = lignes.join("\n").replace(/[<>]/g, "").slice(0, 1500);
    return texte ? J({ ok: true, texte }) : J({ ok: true, vide: true });
  }
  if (route === "/equipe/oubli") {
    const mail = String(body.mail || "").trim().toLowerCase();
    const rep = { ok: true, message: "Si un compte existe avec cette adresse, un e-mail vient de vous être envoyé avec votre lien personnel et un lien pour choisir un nouveau mot de passe. Pensez à regarder dans vos courriers indésirables." };
    if (!validMail(mail)) return J({ erreur: "Merci d'indiquer une adresse e-mail valide." }, 400);
    const c = await eqParMail(env, mail);
    if (c && c.mdp) {
      const base = eqBase(body.base); const exp = Date.now() + 36e5; const t = c.id + "." + exp + "." + await eqSigReinit(env, c, exp);
      await eqEnvoyer(env, c, "Votre lien et votre mot de passe", "Retrouvez votre accès", `<p>Bonjour ${esc(c.prenom)},</p><p>Voici votre lien personnel :</p><p style="word-break:break-all;font-size:12px">${esc(eqLien(base, c))}</p><p>Si vous avez aussi oublié votre mot de passe, appuyez sur le bouton ci-dessous pour en choisir un nouveau (le lien est valable 1 heure).</p><p style="font-size:12px">Si vous n'êtes pas à l'origine de cette demande, ignorez simplement cet e-mail : votre mot de passe actuel reste valable.</p>`, "Choisir un nouveau mot de passe", base + "#reinit=" + encodeURIComponent(t));
    }
    return J(rep);
  }
  if (route === "/equipe/reinit") {
    const [id, exp, sig] = String(body.t || "").split(".");
    if (!id || !exp || !sig || Date.now() > +exp) return J({ erreur: "Ce lien a expiré. Refaites une demande « Mot de passe ou lien oublié »." }, 400);
    if (!mdpFort(body.mdp)) return J({ erreur: MSG_MDP }, 400);
    const c = await eqParId(env, id);
    if (!c.mdp || sig !== await eqSigReinit(env, c, exp)) return J({ erreur: "Ce lien n'est plus valable (déjà utilisé ou expiré). Refaites une demande." }, 400);
    const salt = rnd(16);
    await notion(env, "/pages/" + id, "PATCH", { properties: { "Empreinte mot de passe (ne pas modifier)": T(salt + "|" + await hashPwd(String(body.mdp), salt)) } });
    return J({ ok: true, message: "Votre mot de passe a été changé. Vous pouvez vous connecter." });
  }
  if (route === "/equipe/vente-ajout" || route === "/equipe/ventes" || route === "/equipe/vente-maj") {
    const id = await eqLireJeton(env, body.jeton);
    if (!id) return J({ erreur: "Session expirée : reconnectez-vous." }, 401);
    const c = await eqParId(env, id);
    if (c.archive || !c.mdp) return J({ erreur: "Session expirée : reconnectez-vous." }, 401);
    if (route === "/equipe/ventes") {
      const lignes = await query(env, CFG.DS_VENTES_EQ, { property: "Identifiant conseillère (ne pas modifier)", rich_text: { equals: id } }, [{ property: "Date et heure", direction: "descending" }]);
      return J({ ok: true, ventes: lignes.slice(0, 300).map(eqVenteLue) });
    }
    const r = eqVenteProps(body);
    if (r.erreur) return J({ erreur: r.erreur }, 400);
    if (route === "/equipe/vente-ajout") {
      const pg = await notion(env, "/pages", "POST", { parent: { type: "data_source_id", data_source_id: CFG.DS_VENTES_EQ }, properties: { ...r.props, "Conseillère": T((c.prenom + " " + c.nom).trim()), "Identifiant conseillère (ne pas modifier)": T(id), "Statut": { select: { name: "Enregistrée" } } } });
      return J({ ok: true, vente: eqVenteLue(pg) });
    }
    const vid = String(body.id || "");
    if (!/^[0-9a-f]{8}-?[0-9a-f]{4}-?[0-9a-f]{4}-?[0-9a-f]{4}-?[0-9a-f]{12}$/i.test(vid)) return J({ erreur: "Vente introuvable." }, 400);
    let v; try { v = await notion(env, "/pages/" + vid); } catch (e) { return J({ erreur: "Vente introuvable." }, 404); }
    const ds = v.parent && (v.parent.data_source_id || v.parent.database_id || "");
    if (String(ds).replace(/-/g, "") !== CFG.DS_VENTES_EQ.replace(/-/g, "") || txt(v.properties["Identifiant conseillère (ne pas modifier)"]) !== id) return J({ erreur: "Vente introuvable." }, 404);
    const statut = body.statut === "Annulée" ? "Annulée" : "Enregistrée";
    const maj = await notion(env, "/pages/" + vid, "PATCH", { properties: { ...r.props, "Statut": { select: { name: statut } } } });
    return J({ ok: true, vente: eqVenteLue(maj) });
  }
  return J({ erreur: "Route inconnue." }, 404);
}

async function handle(req, env) {
  const url = new URL(req.url); const route = url.pathname.replace(/\/+$/, "");
  const eqR = route.startsWith("/equipe/");
  const origin = eqR ? (req.headers.get("Origin") === CFG.EQ_ORIGIN ? CFG.EQ_ORIGIN : CFG.ORIGINS[0]) : (CFG.ORIGINS.includes(req.headers.get("Origin")) ? req.headers.get("Origin") : CFG.ORIGINS[0]);
  if (req.method === "OPTIONS") return json({}, 204, origin);
  const J = (o, s) => json(o, s, origin);
  const body = req.method === "POST" ? await req.json().catch(() => ({})) : {};
  const nowD = today(); const auj = iso(nowD);
  if (eqR) return eqRoute(env, route, body, J);

  if (route === "/ping") return J({ ok: true });
  if (route === "/etat-filleules") { try { const r = await query(env, CFG.DS_FILLEULES, { property: "Date d'entrée", date: { is_not_empty: true } }); return J({ ok: true, fiches: r.length }); } catch (e) { return J({ ok: false }); } }

  // Désinscription (lien dans les e-mails)
  if (route === "/desabo") {
    const id = url.searchParams.get("c"), s = url.searchParams.get("s");
    if (!id || s !== await hmac(env, "u|" + id)) return page("Lien invalide", "Ce lien n'est pas valide.");
    await notion(env, "/pages/" + id, "PATCH", { properties: { "Accepte les e-mails": { checkbox: false }, "Ne plus recevoir d'e-mails": { checkbox: true } } });
    return page("C'est fait", "Vous ne recevrez plus d'e-mails de promotion. Vos e-mails de commande continueront d'arriver.");
  }
  // Confirmation d'adresse (lien reçu après inscription)
  if (route === "/confirmer") {
    const id = url.searchParams.get("c"), s = url.searchParams.get("s");
    if (!id || s !== await hmac(env, "v|" + id)) return page("Lien invalide", "Ce lien n'est pas valide.");
    const c = await clientById(env, id);
    if (c.mdp.startsWith("attente|")) await notion(env, "/pages/" + id, "PATCH", { properties: { "Mot de passe chiffré": T(c.mdp.slice(8)) } });
    return page("Compte confirmé", "Votre compte est activé. Vous pouvez maintenant vous connecter sur le site.");
  }

  if (route === "/inscription") {
    const mail = String(body.mail || "").trim().toLowerCase();
    if (!validMail(mail)) return J({ erreur: "Merci d'indiquer une adresse e-mail valide." }, 400);
    if (!mdpFort(body.mdp)) return J({ erreur: MSG_MDP }, 400);
    const prenom = String(body.prenom || "").trim().slice(0, 60), nom = String(body.nom || "").trim().slice(0, 60);
    if (!prenom || !nom) return J({ erreur: "Merci d'indiquer votre prénom et votre nom." }, 400);
    const tel0 = String(body.tel || "").trim(), adr0 = String(body.adresse || "").trim(), cp0 = String(body.cp || "").trim(), ville0 = String(body.ville || "").trim();
    if (!tel0 || !adr0 || !cp0 || !ville0 || !/^\d{4}-\d{2}-\d{2}$/.test(body.naissance || "")) return J({ erreur: "Merci de remplir tous les champs obligatoires (téléphone, adresse, date de naissance)." }, 400);
    if (!body.accepte && !body.newsletter && !body.aucun) return J({ erreur: "Merci de choisir vos préférences d'e-mails." }, 400);
    const salt = rnd(16), h = salt + "|" + await hashPwd(body.mdp, salt);
    const idJeton = body.jeton ? await lireTokenCreation(env, body.jeton) : null;
    let c = idJeton ? await clientById(env, idJeton) : await clientByMail(env, mail);
    if (c && c.mdp && !c.mdp.startsWith("attente|")) return J({ erreur: "Un compte existe déjà avec cette adresse. Connectez-vous." }, 409);
    const props = { "Mot de passe chiffré": T("attente|" + h), "Accepte les e-mails": { checkbox: !!body.accepte }, "Ne plus recevoir d'e-mails": { checkbox: !!body.aucun }, Newsletter: { checkbox: !!body.newsletter }, "Compte site créé le": { date: { start: auj } },
      Adresse: T(adr0.slice(0, 200)), CP: T(cp0.slice(0, 10)), Ville: T(ville0.slice(0, 80)) };
    if (tel0) props["Téléphone"] = { phone_number: tel0.slice(0, 30) };
    if (/^\d{4}-\d{2}-\d{2}$/.test(body.naissance || "")) props["Date de naissance"] = { date: { start: body.naissance } };
    let id;
    if (c) { id = c.id; if (c.mail !== mail) props.Mail = { email: mail }; await notion(env, "/pages/" + id, "PATCH", { properties: props }); }
    else { const pg = await notion(env, "/pages", "POST", { parent: { type: "data_source_id", data_source_id: CFG.DS_CLIENTS }, properties: { ...props, "Client(e)": TT(prenom + " " + nom.toUpperCase()), Mail: { email: mail }, "Points fidélité": { number: 0 } } }); id = pg.id; }
    const lien = `${url.origin}/confirmer?c=${id}&s=${encodeURIComponent(await hmac(env, "v|" + id))}`;
    await brevo(env, mail, "Confirmez votre compte", simple("Bienvenue !", `<p>Bonjour ${esc(prenom)},</p><p>Merci d'avoir créé votre compte. Pour l'activer, appuyez sur le bouton ci-dessous.</p>`).replace("</table></body>", bouton("Confirmer mon compte", lien) + pied() + "</table></body>"));
    return J({ ok: true, message: "Un e-mail de confirmation vient de vous être envoyé. Appuyez sur le lien qu'il contient, puis connectez-vous." });
  }

  if (route === "/connexion") {
    const mail = String(body.mail || "").trim().toLowerCase(); const c = mail ? await clientByMail(env, mail) : null;
    const [salt, h] = c && c.mdp && !c.mdp.startsWith("attente|") ? c.mdp.split("|") : [];
    if (!c || !salt || (h !== await hashPwd(String(body.mdp || ""), salt) && h !== await hashPwd(String(body.mdp || "").trim(), salt))) {
      if (c && c.mdp.startsWith("attente|")) return J({ erreur: "Confirmez d'abord votre adresse avec le lien reçu par e-mail." }, 401);
      return J({ erreur: "Adresse ou mot de passe incorrect." }, 401);
    }
    return J({ token: await makeToken(env, c.id), profil: await profil(env, c) });
  }

  // Mot de passe oublié : envoi d'un lien valable 1 heure
  if (route === "/mot-de-passe-oublie") {
    const mail = String(body.mail || "").trim().toLowerCase();
    const rep = { ok: true, message: "Si un compte existe avec cette adresse, un e-mail vient de vous être envoyé avec un lien pour choisir un nouveau mot de passe. Pensez à regarder dans vos courriers indésirables." };
    if (!validMail(mail)) return J({ erreur: "Merci d'indiquer une adresse e-mail valide." }, 400);
    const c = await clientByMail(env, mail);
    if (c && c.mdp && !c.mdp.startsWith("attente|")) {
      const exp = Date.now() + 36e5; const t = c.id + "." + exp + "." + await sigReinit(env, c, exp);
      const lien = CFG.SITE + "/#reinit=" + encodeURIComponent(t);
      await brevo(env, mail, "Réinitialisez votre mot de passe", simple("Nouveau mot de passe", `<p>Bonjour ${esc(c.prenom)},</p><p>Vous avez demandé à changer votre mot de passe. Appuyez sur le bouton ci-dessous pour en choisir un nouveau (le lien est valable 1 heure).</p><p style="font-size:12px">Si vous n'êtes pas à l'origine de cette demande, ignorez simplement cet e-mail : votre mot de passe actuel reste valable.</p>`).replace("</table></body>", bouton("Choisir un nouveau mot de passe", lien) + pied() + "</table></body>"));
    }
    return J(rep);
  }
  if (route === "/reinitialiser") {
    const [id, exp, sig] = String(body.t || "").split(".");
    if (!id || !exp || !sig || Date.now() > +exp) return J({ erreur: "Ce lien a expiré. Refaites une demande de mot de passe oublié." }, 400);
    if (!mdpFort(body.mdp)) return J({ erreur: MSG_MDP }, 400);
    const c = await clientById(env, id);
    if (!c.mdp || c.mdp.startsWith("attente|") || sig !== await sigReinit(env, c, exp)) return J({ erreur: "Ce lien n'est plus valable (déjà utilisé ou expiré). Refaites une demande." }, 400);
    const salt = rnd(16);
    await notion(env, "/pages/" + id, "PATCH", { properties: { "Mot de passe chiffré": T(salt + "|" + await hashPwd(body.mdp, salt)) } });
    return J({ ok: true, message: "Votre mot de passe a été changé. Vous pouvez vous connecter." });
  }

  // Meilleures ventes (commandes passées sur le site) : nombre d'exemplaires par nom de produit
  if (route === "/meilleures-ventes") {
    const cmds = await query(env, CFG.DS_COMMANDES, { property: "Origine", select: { equals: "Site" } });
    const noms = {};
    for (const pg of cmds) for (const l of txt(pg.properties.Produits).split("\n")) {
      const m = l.match(/^(\d+) × (.+?)(?: \(.*\))? — /); if (m) noms[m[2].trim()] = (noms[m[2].trim()] || 0) + (+m[1]);
    }
    return J({ noms });
  }

  // --- Routes protégées ---
  const authId = await readToken(env, req);
  if (route === "/moi") { if (!authId) return J({ erreur: "Veuillez vous reconnecter." }, 401); return J({ profil: await profil(env, await clientById(env, authId)) }); }
  if (route === "/panier") {
    if (!authId) return J({ erreur: "Veuillez vous reconnecter." }, 401);
    const s = JSON.stringify(Array.isArray(body.panier) ? body.panier.slice(0, 60) : []);
    if (s.length > 1990) return J({ erreur: "Panier trop volumineux." }, 400);
    await notion(env, "/pages/" + authId, "PATCH", { properties: { "Panier en cours": T(s) } }); return J({ ok: true });
  }
  if (route === "/preferences") {
    if (!authId) return J({ erreur: "Veuillez vous reconnecter." }, 401);
    await notion(env, "/pages/" + authId, "PATCH", { properties: { "Accepte les e-mails": { checkbox: !!body.accepte }, Newsletter: { checkbox: !!body.newsletter }, "Ne plus recevoir d'e-mails": { checkbox: !body.accepte && !body.newsletter } } }); return J({ ok: true });
  }
  if (route === "/supprimer-compte") {
    if (!authId) return J({ erreur: "Veuillez vous reconnecter." }, 401);
    const c = await clientById(env, authId); const [salt, h] = c.mdp.split("|");
    if (h !== await hashPwd(String(body.mdp || ""), salt)) return J({ erreur: "Mot de passe incorrect." }, 401);
    await notion(env, "/pages/" + authId, "PATCH", { properties: { "Mot de passe chiffré": T(""), "Panier en cours": T(""), "Accepte les e-mails": { checkbox: false }, "Compte site créé le": { date: null } } });
    return J({ ok: true });
  }
  if (route === "/changer-mdp") {
    if (!authId) return J({ erreur: "Veuillez vous reconnecter." }, 401);
    const c = await clientById(env, authId);
    if (!await mdpOk(c, body.ancien)) return J({ erreur: "Votre mot de passe actuel est incorrect." }, 401);
    if (!mdpFort(body.nouveau)) return J({ erreur: MSG_MDP }, 400);
    const salt = rnd(16);
    await notion(env, "/pages/" + authId, "PATCH", { properties: { "Mot de passe chiffré": T(salt + "|" + await hashPwd(body.nouveau, salt)) } });
    return J({ ok: true });
  }
  if (route === "/changer-mail") {
    if (!authId) return J({ erreur: "Veuillez vous reconnecter." }, 401);
    const c = await clientById(env, authId); const mail = String(body.mail || "").trim().toLowerCase();
    if (!validMail(mail)) return J({ erreur: "Merci d'indiquer une adresse e-mail valide." }, 400);
    if (!await mdpOk(c, body.mdp)) return J({ erreur: "Mot de passe incorrect." }, 401);
    const autre = await clientByMail(env, mail);
    if (autre && autre.id !== authId) return J({ erreur: "Cette adresse est déjà utilisée par un autre compte." }, 409);
    await notion(env, "/pages/" + authId, "PATCH", { properties: { Mail: { email: mail } } });
    return J({ ok: true, profil: await profil(env, await clientById(env, authId)) });
  }
  if (route === "/modifier-profil") {
    if (!authId) return J({ erreur: "Veuillez vous reconnecter." }, 401);
    const props = {};
    if (body.prenom !== undefined || body.nom !== undefined) {
      const pr = String(body.prenom || "").trim().slice(0, 60), no = String(body.nom || "").trim().slice(0, 60);
      if (!pr || !no) return J({ erreur: "Le prénom et le nom ne peuvent pas être vides." }, 400);
      props["Client(e)"] = TT(pr + " " + no.toUpperCase());
    }

    const o = { numero, lignes, totalProduits, remise, port, total, mode, relais: body.relais, adresse: body.adresse };
    const info = `<p style="margin:12px 0 0;font-size:13px">Vous avez gagné <b>${pts} point${pts > 1 ? "s" : ""}</b> de fidélité${cout ? " (" + cout + " points utilisés pour votre bon)" : ""}.<br>Solde de vos points de fidélité : <b>${nouveaux} point${nouveaux > 1 ? "s" : ""}</b>.${crees.length ? "<br>Un nouveau bon fidélité vous attend dans votre compte : " + crees.map((x) => "<b>" + x.montant + " €</b> (code " + esc(x.code) + ")").join(", ") + "." : ""}</p>`;
    await brevo(env, mail, "Votre commande " + numero, mailConfirmation(c.prenom || "", o, info));
    await brevo(env, CFG.ZULAY, "Nouvelle commande " + numero, simple("Nouvelle commande", `<p><b>${esc(c.nom)}</b> — ${esc(mail)} ${esc(c.tel || "")}</p>${lignesCommande(o)}<p>${livraisonTexte(o)}</p>${body.note ? "<p>Note : " + esc(body.note) + "</p>" : ""}`));
    return J({ ok: true, numero, points: pts, total: nouveaux, bonsCrees: crees.map((x) => x.code) });
  }

  // --- Carte cadeau (achat sur le site, PDF téléchargé par l'acheteur) ---
  if (route === "/carte-cadeau/creer") {
    const montant = Math.round(+body.montant || 0); if (montant < 5 || montant > 500) return J({ erreur: "Montant entre 5 € et 500 €." }, 400);
    const mail = String(body.mail || "").trim().toLowerCase(); if (!validMail(mail)) return J({ erreur: "Merci d'indiquer votre adresse e-mail." }, 400);
    const fin = (() => { const d = new Date(auj + "T12:00:00"); d.setFullYear(d.getFullYear() + 1); return iso(d); })();
    const b = await creerBon(env, { type: "Carte cadeau", montant, fin, pour: String(body.pour || "").slice(0, 60), de: String(body.de || "").slice(0, 60), message: String(body.message || "").slice(0, 160), mailDest: mail, payee: false });
    await brevo(env, CFG.ZULAY, "Carte cadeau à régler : " + b.code, simple("Nouvelle carte cadeau", `<p>Code <b>${b.code}</b> — ${montant} € pour <b>${esc(body.pour)}</b>, de la part de ${esc(body.de)}. Acheteur : ${esc(mail)}.</p><p>Quand le paiement est reçu, coche « Payée » dans Notion (Bons et cartes cadeaux).</p>`));
    return J({ ok: true, code: b.code });
  }
  if (route === "/carte-cadeau/lire") {
    const b = await bonParCode(env, body.code); const mail = String(body.mail || "").trim().toLowerCase();
    if (!b || b.type !== "Carte cadeau") return J({ erreur: "Code inconnu." }, 404);
    const p = await notion(env, "/pages/" + b.id); const dest = ((p.properties["Destinataire e-mail"] || {}).email || "").toLowerCase();
    if (dest !== mail) return J({ erreur: "Cette adresse ne correspond pas à l'achat." }, 403);
    if (!b.payee) return J({ attente: true, message: "Votre carte cadeau sera disponible dès réception du paiement." });
    return J({ carte: { code: b.code, montant: b.montant, pour: b.pour, de: b.de, message: b.message, fin: b.fin, utilise: b.utilise } });
  }
  return J({ erreur: "Introuvable" }, 404);
}

async function tacheAvis(env, base) {
  const r = await query(env, CFG.DS_COMMANDES, { and: [{ property: "Statut", select: { equals: "Expédiée" } }, { property: "Origine", select: { equals: "Site" } }] });

  }

}
async function tacheAnniversaires(env, base) {
  const d = today(); const auj = iso(d); if (auj < CFG.ANNIV_DEBUT) return { mails: 0, sms: [] };
  const jours = new Set(); for (let k = 0; k <= CFG.ANNIV_RATTRAPAGE; k++) { const j = addDays(auj, -k); if (j >= CFG.ANNIV_DEBUT) jours.add(j.slice(5)); }
  const tous = (await query(env, CFG.DS_CLIENTS, { property: "Date de naissance", date: { is_not_empty: true } })).map(client);
  const fin = addDays(auj, CFG.ANNIV_JOURS); const sms = []; let mails = 0;
  const envoyer = async (c, code, finBon) => {
    const des = `${base}/desabo?c=${c.id}&s=${encodeURIComponent(await hmac(env, "u|" + c.id))}`;
    await brevo(env, c.mail, "Joyeux anniversaire " + c.prenom + " !", mailAnniv(c.prenom, code, finBon.split("-").reverse().join("/"), des), { headers: { "List-Unsubscribe": `<${des}>` } });

    if (mailOk(c)) {

      sms.push({ nom: c.nom, tel: c.tel, texte: smsAnniv(c.prenom, b.code, CFG.SITE + "/#compte"), objet: "Anniversaire" });
    }
  }
  // Rattrapage : cartes déjà créées mais jamais envoyées par e-mail (adresse corrigée, case non cochée…)

      catch (e) { console.log("Anniversaire rattrapage", e.message); }
    }
  } catch (e) { console.log("Rattrapage anniversaires", e.message); }

async function tacheFidelite(env, base) {

async function tacheExh(env) {
  let envoyes = 0;
  const cmds = await query(env, CFG.DS_EXH_COMMANDES, { or: [{ property: "Statut", select: { equals: "Validée" } }, { property: "Statut", select: { equals: "En cours de commande" } }, { property: "Statut", select: { equals: "À récupérer" } }, { property: "Statut", select: { equals: "Expédiée" } }] });
  for (const pg of cmds) {
    const p = pg.properties, statut = sel(p["Statut"]); if (!statut || txt(p["Statut notifié"]) === statut) continue;
    const mail = (p["Mail"] || {}).email || "", numero = txt(p["Commande"]), prenom = txt(p["Prénom"]), mot = txt(p["Mot pour la cliente"]);
    if (!validMail(mail)) continue;
    try {
      if (statut === "Validée") {
        const delai = txt(p["Délai estimé"]); if (!delai) continue;
        const totalProduits = num(p["Total produits"]), port = num(p["Frais de port"]); let info = "";
        let pts = (p["Points gagnés"] || {}).number;
        const c = await clientByMail(env, mail);
        if (typeof pts !== "number") {
          pts = Math.floor(totalProduits);
          if (c) { const nouveaux = c.points + pts; await notion(env, "/pages/" + c.id, "PATCH", { properties: { "Points fidélité": { number: nouveaux }, "Dernière commande": { date: { start: iso(today()) } } } }); c.points = nouveaux; await verifierPaliers(env, c, nouveaux); }
          await notion(env, "/pages/" + pg.id, "PATCH", { properties: { "Points gagnés": { number: pts } } });
        }
        if (c) info = `<p style="margin:12px 0 0;font-size:13px">Vous gagnez <b>${pts} point${pts > 1 ? "s" : ""}</b> de fidélité : il vous en reste <b>${c.points}</b>. 1 € dépensé, hors frais de port, = 1 point, et vos points sont gardés à vie tant que vous ne les utilisez pas.</p>`;
        else info = `<p style="margin:12px 0 0;font-size:13px">Créez votre compte pour cumuler vos points fidélité : 1 € dépensé, hors frais de port, = 1 point, gardé à vie tant que vous ne l'utilisez pas.</p>` + `<p style="margin:8px 0 0"><a href="${esc(CFG.SITE + "/#compte")}" style="color:${C.PL}">Créer mon compte</a></p>`;
        await brevo(env, mail, "Merci pour votre commande " + numero, mailExhValidee(prenom, numero, txt(p["Produits"]), { totalProduits, port, total: num(p["Total"]), livraison: txt(p["Livraison"]) }, sel(p["Réglé par"]), mot, info, delai));
      } else if (statut === "En cours de commande") {
        await brevo(env, mail, "Votre commande " + numero + " est en cours", mailExhEnCours(prenom, numero, txt(p["Délai estimé"]), mot));
      } else if (statut === "Expédiée") {
        const suivi = txt(p["N° de suivi"]), lien0 = (p["Lien de suivi"] || {}).url || ""; if (!suivi && !lien0) continue;
        const lien = lien0 || lienChrono(suivi);
        await brevo(env, mail, "Votre commande " + numero + " est expédiée", mailExhExpediee(prenom, numero, suivi, lien, txt(p["Livraison"]), mot));
      } else await brevo(env, mail, "Votre commande EXHALESSENCE est prête", mailExhRecuperer(prenom, numero, mot));
    } catch (e) { console.log("Commande EXH", e.message); continue; }
    await notion(env, "/pages/" + pg.id, "PATCH", { properties: { "Statut notifié": T(statut) } }); envoyes++;
  }
  const res = await query(env, CFG.DS_EXH_RESERVATIONS, { property: "Statut", select: { equals: "Validée" } });
  for (const pg of res) {
    const p = pg.properties; if (txt(p["Statut notifié"]) === "Validée") continue;
    const mail = (p["Mail"] || {}).email || ""; if (!validMail(mail)) continue;
    try { await brevo(env, mail, "Votre réservation est validée : " + txt(p["Prestation"]), mailExhReservation(txt(p["Prénom"]), txt(p["Prestation"]), txt(p["Mot pour la cliente"]))); }
    catch (e) { console.log("Réservation EXH", e.message); continue; }
    await notion(env, "/pages/" + pg.id, "PATCH", { properties: { "Statut notifié": T("Validée") } }); envoyes++;
  }
  return envoyes;
}
async function quotidien(env, base) {
  const a = await tacheAnniversaires(env, base), f = await tacheFidelite(env, base), v = await tacheAvis(env, base);
  let fl = { envoyes: 0, restes: [] }; try { fl = await tacheFilleules(env); } catch (e) { console.log("Filleules", e.message); }
  await resumeMatin(env, [...a.sms, ...f.sms, ...(fl.restes || [])]); return { anniv: a.mails, fidelite: f.mails, avis: v, filleules: fl.envoyes, sms: a.sms.length + f.sms.length };
}

  async fetch(req, env) {
    try { return await handle(req, env); }
    catch (e) {
      console.log("Erreur", e.message);
      const o = req.headers.get("Origin"), origin = (CFG.ORIGINS.includes(o) || (o === CFG.EQ_ORIGIN && new URL(req.url).pathname.startsWith("/equipe/"))) ? o : CFG.ORIGINS[0];
      const app = new URL(req.url).pathname.startsWith("/app/");
      return json({ erreur: app ? "Notion : " + String(e.message).slice(0, 220) : "Un souci est survenu. Réessayez dans un instant." }, app ? 502 : 500, origin);
    }
  },
  async scheduled(evt, env, ctx) {
    const base = env.BASE_URL || "https://aphrodite-relais.les-essences-parfumees-d-aphrodite.workers.dev";
    ctx.waitUntil((async () => {
      await tacheStatuts(env);
      try { await tacheMessages(env); } catch (e) { console.log("Messages", e.message); }
      try { await tacheExh(env); } catch (e) { console.log("Exh", e.message); }
      const h = today(); if (h.getHours() === CFG.HEURE_QUOTIDIENNE && h.getMinutes() < 15) await quotidien(env, base);
    })());
  },
